cmake_minimum_required(VERSION 3.15)
project(correctionlib_native_plugin_tests LANGUAGES CXX)
if(NOT TARGET correctionlib)
  find_package(correctionlib CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
enable_testing()
add_library(test_node_provider SHARED ../../examples/node_plugin/nodes.cc)
target_link_libraries(test_node_provider PRIVATE correctionlib)
set_target_properties(test_node_provider PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
foreach(kind IN ITEMS wrong_api wrong_build failing)
  add_library(${kind}_provider SHARED bad_provider.cc)
  target_compile_definitions(${kind}_provider PRIVATE ${kind}=1)
  target_link_libraries(${kind}_provider PRIVATE correctionlib)
  set_target_properties(${kind}_provider PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endforeach()
add_executable(test_native_plugins test_native_plugins.cc)
target_link_libraries(test_native_plugins PRIVATE correctionlib Threads::Threads)
add_test(NAME native_plugins COMMAND test_native_plugins $<TARGET_FILE:test_node_provider>
  $<TARGET_FILE:wrong_api_provider> $<TARGET_FILE:failing_provider> $<TARGET_FILE:wrong_build_provider>)
add_executable(test_linked_plugin test_linked_plugin.cc)
target_link_libraries(test_linked_plugin PRIVATE correctionlib test_node_provider)
add_test(NAME linked_plugin COMMAND test_linked_plugin)
